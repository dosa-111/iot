Fingerprint sensor (R307) module pin number    |   USB to TTL converter pins

1  Vcc                                        |   3V / 5V
2  Gnd                                        |   Gnd
3  Tx                                         |   Rx
4  Rx                                         |   Tx

 COMMANDS :
- sudo raspi- config (Enable serial port)

- pip3 install adafruit- circuitpython-fingerprint

- ls /dev/ttyUSB* 


import time
import serial
import adafruit_fingerprint

uart = serial.Serial("/dev/ttyUSB0", baudrate=57600)
finger = adafruit_fingerprint.Adafruit_Fingerprint(uart)

while True:
    print("\n1. Enroll")
    print("2. Search")
    print("3. Delete")

    choice = input("Enter choice: ")

    if choice == "1":
        id = int(input("Enter ID (1-127): "))

        for i in [1, 2]:
            print("Place finger", i)
            while finger.get_image() != adafruit_fingerprint.OK:
                time.sleep(0.1)
            finger.image_2_tz(i)

            if i == 1:
                print("Remove finger")
                while finger.get_image() != adafruit_fingerprint.NOFINGER:
                    time.sleep(0.1)

        if finger.create_model() == adafruit_fingerprint.OK:
            if finger.store_model(id) == adafruit_fingerprint.OK:
                print("Fingerprint Stored!")
            else:
                print("Storage Failed!")
        else:
            print("Enrollment Failed!")

    elif choice == "2":
        print("Place finger")
        while finger.get_image() != adafruit_fingerprint.OK:
            time.sleep(0.1)

        finger.image_2_tz(1)

        if finger.finger_search() == adafruit_fingerprint.OK:
            print("Fingerprint Found! ID:", finger.finger_id)
        else:
            print("Fingerprint Not Found!")

    elif choice == "3":
        id = int(input("Enter ID: "))
        if finger.delete_model(id) == adafruit_fingerprint.OK:
            print("Fingerprint Deleted!")
        else:
            print("Delete Failed!")
